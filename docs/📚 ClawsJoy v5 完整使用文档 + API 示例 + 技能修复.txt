四、Python SDK 完整示例
python
#!/usr/bin/env python3
"""ClawsJoy v5 Python SDK 示例"""

import requests
import json

class ClawsJoyClient:
    """ClawsJoy v5 客户端"""
    
    def __init__(self, base_url="http://127.0.0.1:5002"):
        self.base_url = base_url
        self.user_id = "default_user"
    
    def chat(self, message: str, user_id: str = None) -> dict:
        """发送聊天消息"""
        url = f"{self.base_url}/api/v5/enhanced/chat"
        payload = {
            "user_id": user_id or self.user_id,
            "message": message
        }
        response = requests.post(url, json=payload)
        return response.json()
    
    def remember_fact(self, fact: str, user_id: str = None) -> dict:
        """存储事实记忆"""
        url = f"{self.base_url}/api/v5/memory/remember"
        payload = {
            "user_id": user_id or self.user_id,
            "fact": fact
        }
        return requests.post(url, json=payload).json()
    
    def remember_preference(self, key: str, value: str, user_id: str = None) -> dict:
        """存储用户偏好"""
        url = f"{self.base_url}/api/v5/memory/remember"
        payload = {
            "user_id": user_id or self.user_id,
            "preference": {key: value}
        }
        return requests.post(url, json=payload).json()
    
    def recall_memory(self, limit: int = 10, user_id: str = None) -> dict:
        """检索记忆"""
        url = f"{self.base_url}/api/v5/memory/recall"
        payload = {
            "user_id": user_id or self.user_id,
            "limit": limit
        }
        return requests.post(url, json=payload).json()
    
    def health_check(self) -> dict:
        """健康检查"""
        return requests.get(f"{self.base_url}/health").json()
    
    def get_metrics(self) -> dict:
        """获取监控指标"""
        return requests.get(f"{self.base_url}/metrics").json()


# 使用示例
if __name__ == "__main__":
    client = ClawsJoyClient()
    
    # 1. 健康检查
    print("=== 健康检查 ===")
    print(client.health_check())
    
    # 2. 聊天
    print("\n=== 聊天测试 ===")
    print(client.chat("你好"))
    print(client.chat("今天天气怎么样"))
    print(client.chat("1+2等于多少"))
    print(client.chat("用粤语说你好"))
    
    # 3. 记忆系统
    print("\n=== 记忆系统 ===")
    client.remember_fact("用户喜欢简洁风格")
    client.remember_preference("theme", "dark")
    memories = client.recall_memory(limit=5)
    print(f"记忆数量: {len(memories.get('results', []))}")
    
    # 4. 监控
    print("\n=== 监控指标 ===")
    metrics = client.get_metrics()
    print(f"CPU: {metrics.get('cpu_percent')}%")
    print(f"内存: {metrics.get('memory_percent')}%")

#!/usr/bin/env python3
"""技能执行修复示例"""

import sys
sys.path.insert(0, '/home/flybo/clawsjoy_v5')

from core.lib.skill_loader_v3 import skill_loader

def execute_skill_safe(skill_name: str, params: dict = None):
    """安全执行技能"""
    try:
        # 获取技能信息
        skill_info = skill_loader.get_skill(skill_name)
        if not skill_info:
            return {"error": f"技能 {skill_name} 不存在"}
        
        # 执行技能
        result = skill_loader.execute(skill_name, params or {})
        return {"success": True, "result": result}
    except Exception as e:
        return {"success": False, "error": str(e)}

# 测试执行
if __name__ == "__main__":
    # 注意：实际的技能名称和参数需要根据技能定义调整
    test_skills = [
        ("math", {"expression": "2+2"}),
        ("calculator", {"a": 10, "b": 5, "op": "+"}),
        ("get_weather", {"city": "北京"}),
    ]
    
    for skill_name, params in test_skills:
        result = execute_skill_safe(skill_name, params)
        print(f"{skill_name}: {result}")

"""WebSocket 语音识别示例"""

import asyncio
import websockets
import json

async def voice_recognition():
    uri = "ws://localhost:5003/ws/voice"
    async with websockets.connect(uri) as websocket:
        # 发送音频数据
        with open("test_audio.wav", "rb") as f:
            audio_data = f.read()
            await websocket.send(audio_data)
        
        # 接收识别结果
        response = await websocket.recv()
        result = json.loads(response)
        print(f"识别结果: {result}")
